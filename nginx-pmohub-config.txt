# ========================================
# CONFIGURAÇÕES A ADICIONAR NO NGINX
# Arquivo: /etc/nginx/sites-available/interno-sandech
# Posição: Após a seção /api/reports (linha 115) e antes de /worklocation
# ========================================

    # ========================================
    # Location: /pmohub/api → Hub PMO API (porta 7000)
    # ========================================
    location /pmohub/api/ {
        rewrite ^/pmohub/api/(.*) /api/$1 break;
        proxy_pass http://localhost:7000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
    }

    # ========================================
    # Location: /pmohub → Hub PMO Frontend (porta 7001)
    # ========================================
    location /pmohub {
        return 301 https://$server_name/pmohub/;
    }

    location /pmohub/ {
        proxy_pass http://localhost:7001;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_connect_timeout 300;
        proxy_send_timeout 300;
        proxy_read_timeout 300;
    }

# ========================================
# INSTRUÇÕES DE INSTALAÇÃO:
# ========================================

sudo cp /etc/nginx/sites-available/interno-sandech /etc/nginx/sites-available/interno-sandech.backup

# 2. Editar o arquivo:
sudo nano /etc/nginx/sites-available/interno-sandech

# 3. Adicionar o conteúdo acima após a linha 115 (após /api/reports)

# 4. Validar sintaxe:
sudo nginx -t

# 5. Recarregar Nginx:
sudo systemctl reload nginx
